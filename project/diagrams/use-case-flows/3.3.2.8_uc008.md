flowchart TD

    classDef startEnd fill:#f9f,stroke:#333,stroke-width:2px
    classDef action fill:#cff,stroke:#333
    classDef decision fill:#fc9,stroke:#333

        D1[Thêm]:::action
        D2[Sửa]:::action
        D3[Xóa]:::action

    D --> thaoTac

    H -- Có --> I[Lưu thay đổi vào CSDL]:::process

    J --> K[Tiếp tục thao tác hoặc thoát]:::action
    K --> Z([End]):::startEnd

    H -- Không --> M[Hiển thị lỗi nhập liệu]:::process
